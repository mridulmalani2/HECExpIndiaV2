Exact CSS to add:

/* Frosted glass cards (light/dark aware) /
:root {
--glass-blur: 12px; / moderate blur for good performance */
--glass-bg-light: rgba(255, 255, 255, 0.16);
--glass-bg-dark: rgba(32, 33, 42, 0.30);
--glass-border-light: rgba(255, 255, 255, 0.28);
--glass-border-dark: rgba(255, 255, 255, 0.12);
--glass-shadow-light: 0 4px 20px rgba(0, 0, 0, 0.10);
--glass-shadow-dark: 0 4px 20px rgba(0, 0, 0, 0.30);
}

/* Base (light mode by default) */
.card {
background-color: var(--glass-bg-light);
backdrop-filter: blur(var(--glass-blur));
-webkit-backdrop-filter: blur(var(--glass-blur));
border: 1px solid var(--glass-border-light);
box-shadow: var(--glass-shadow-light);
border-radius: 16px;
transition: transform .25s ease, box-shadow .25s ease, background-color .25s ease, border-color .25s ease;
}

/* Hover (subtle lift) */
.card:hover {
transform: translateY(-3px);
box-shadow: 0 6px 26px rgba(0, 0, 0, 0.15);
}

/* Dark mode */
@media (prefers-color-scheme: dark) {
.card {
background-color: var(--glass-bg-dark);
border-color: var(--glass-border-dark);
box-shadow: var(--glass-shadow-dark);
}
}

/* Fallback for browsers without backdrop-filter */
@supports not ((backdrop-filter: blur(1px)) or (-webkit-backdrop-filter: blur(1px))) {
.card {
background-color: rgba(255, 255, 255, 0.85);
}
@media (prefers-color-scheme: dark) {
.card { background-color: rgba(32, 33, 42, 0.85); }
}
}

/* Keep inner thumbnails/text unaffected */
.card img { display:block; }
